MySQL (XAMPP)

Sintaxe usada no phpMyAdmin / linha de comando do XAMPP

MySQL 
-- Criando a tabela
CREATE TABLE aluno (
    id_aluno INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE
);

-- Inserindo dados
INSERT INTO aluno (nome, email)
VALUES ('Maria Silva', 'maria@etec.sp.gov.br');

-- Consulta com JOIN e LIMIT
SELECT a.nome, t.nome AS turma
FROM aluno a
JOIN turma t ON a.id_turma = t.id_turma
LIMIT 10;

SQL Server (T-SQL)

Sintaxe usada no SQL Server Management Studio (SSMS)

SQL Server 
-- Criando a tabela
CREATE TABLE aluno (
    id_aluno INT IDENTITY(1,1) PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150) UNIQUE
);

-- Inserindo dados
INSERT INTO aluno (nome, email)
VALUES ('Maria Silva', 'maria@etec.sp.gov.br');

-- Consulta com JOIN e TOP
SELECT TOP 10 a.nome, t.nome AS turma
FROM aluno a
JOIN turma t ON a.id_turma = t.id_turma;
